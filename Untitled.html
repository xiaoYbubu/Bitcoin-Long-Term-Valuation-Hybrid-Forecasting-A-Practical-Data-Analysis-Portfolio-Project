import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.api as sm
import tkinter as tk
from tkinter import filedialog

# 1. 弹出窗口让你选择 CSV 文件（彻底解决 FileNotFoundError）
root = tk.Tk()
root.withdraw()
root.attributes('-topmost', True)
print("请在弹出的窗口中选择你的比特币 CSV 文件...")
file_path = filedialog.askopenfilename(
    title="选择 Bitcoin 数据表格",
    filetypes=[("CSV Files", "*.csv"), ("Excel Files", "*.xlsx;*.xls"), ("All Files", "*.*")]
)

if not file_path:
    print("未选择文件，程序终止。")
else:
    print(f"已成功加载文件: {file_path}")
    
    # 根据文件类型读取数据
    if file_path.endswith('.csv'):
        df = pd.read_csv(file_path)
    else:
        df = pd.read_excel(file_path)

    # 2. 数据清洗（解决之前报错的类型与格式问题）
    df['Date'] = pd.to_datetime(df['Date'])
    df = df.sort_values('Date').reset_index(drop=True)
    
    # 清理 Price 列：去除逗号并转为浮点数
    df['Price'] = df['Price'].astype(str).str.replace(',', '').astype(float)

    # 3. 特征工程：计算创世日天数及其对数值
    genesis_date = pd.to_datetime('2009-01-09')
    df['Days'] = (df['Date'] - genesis_date).dt.days
    df['log_Days'] = np.log(df['Days'])
    df['log_Price'] = np.log(df['Price'])

    # 4. 构建 OLS 回归模型
    X = sm.add_constant(df['log_Days'])
    y = df['log_Price']
    model = sm.OLS(y, X).fit()

    print("\n================ 模型拟合统计摘要 ================")
    print(model.summary())

    # 5. 生成未来日期序列（预测至 2028 年底）
    future_dates = pd.date_range(start='2017-01-01', end='2028-12-31', freq='D')
    future_df = pd.DataFrame({'Date': future_dates})
    future_df['Days'] = (future_df['Date'] - genesis_date).dt.days
    future_df['log_Days'] = np.log(future_df['Days'])

    # 6. 计算预测值与 95% 置信区间
    pred_X = sm.add_constant(future_df['log_Days'])
    predictions = model.get_prediction(pred_X).summary_frame(alpha=0.05)

    future_df['Pred_Price'] = np.exp(predictions['mean'])
    future_df['Lower_Bound'] = np.exp(predictions['obs_ci_lower']) # 95% 下限
    future_df['Upper_Bound'] = np.exp(predictions['obs_ci_upper']) # 95% 上限

    # 7. 定义查询函数
    def predict_price(query_date_str):
        q_date = pd.to_datetime(query_date_str)
        res = future_df[future_df['Date'] == q_date]
        if len(res) == 0:
            print(f"[{query_date_str}] 日期超出预测范围")
            return
        p = res['Pred_Price'].values[0]
        low = res['Lower_Bound'].values[0]
        high = res['Upper_Bound'].values[0]
        print(f"[{query_date_str}] 预测期望价格: ${p:,.2f} | 95% 区间: [${low:,.2f} ~ ${high:,.2f}]")

    print("\n================ 未来任意日期预测测试 ================")
    predict_price('2026-10-01')
    predict_price('2027-06-18')

    # 8. 绘制拟合趋势与置信区间通道图
    plt.figure(figsize=(12, 6))
    plt.plot(df['Date'], df['Price'], label='Actual BTC Price', color='orange', alpha=0.8)
    plt.plot(future_df['Date'], future_df['Pred_Price'], label='Trend Line (Power Law)', color='blue', linestyle='--')
    plt.fill_between(future_df['Date'], future_df['Lower_Bound'], future_df['Upper_Bound'], 
                     color='blue', alpha=0.15, label='95% Prediction Band')
    plt.yscale('log')
    plt.title('Bitcoin Long-Term Power Law Trend & Prediction Interval', fontsize=14)
    plt.xlabel('Date')
    plt.ylabel('Price (USD, Log Scale)')
    plt.legend()
    plt.grid(True, which="both", ls="--", alpha=0.5)
    plt.show()

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import statsmodels.api as sm

# 1. 提取微观量价因子并清理数据（修复 Pandas 2.1+ 兼容性与百分号问题）
# 清理百分号和逗号（如果存在）
for col in ['Daily Return', '30-Day Volatility', 'logvolume']:
    if col in df.columns:
        df[col] = df[col].astype(str).str.replace('%', '').str.replace(',', '')
        df[col] = pd.to_numeric(df[col], errors='coerce')

# 填充缺失值（使用新版 .bfill() 方法）
df['Vol_30D'] = df['30-Day Volatility'].bfill().ffill()
df['Log_Vol'] = df['logvolume'].bfill().ffill()
df['Daily_Ret'] = df['Daily Return'].fillna(0)

# 构建滞后特征
df['Ret_Lag1'] = df['Daily_Ret'].shift(1).fillna(0)
df['Ret_Lag2'] = df['Daily_Ret'].shift(2).fillna(0)

# 2. 构建残差短期预测模型 (Predicting Residual Dynamics)
df['Z_diff'] = df['Valuation_ZScore'].diff()
df_model = df.dropna().copy()

X_micro = df_model[['Valuation_ZScore', 'Vol_30D', 'Log_Vol', 'Ret_Lag1', 'Ret_Lag2']]
X_micro = sm.add_constant(X_micro)
y_micro = df_model['Z_diff']

# 拟合微观修正回归模型
micro_model = sm.OLS(y_micro, X_micro).fit()

print("================ 短期微观因子修正模型摘要 ================")
print(micro_model.summary())

# 3. 推演 2026-09-04 至 2026-12-31 的动态预测
last_known_date = df['Date'].max() # 2026-09-03
future_projection_dates = pd.date_range(start='2026-09-04', end='2026-12-31', freq='D')

# 获取最新的微观状态
latest_z = df.iloc[-1]['Valuation_ZScore']
latest_vol = df.iloc[-1]['Vol_30D']
latest_logvol = df.iloc[-1]['Log_Vol']

projected_z = []
current_z = latest_z

# 递归推演未来残差走势（均值回归过程）
for d in future_projection_dates:
    pred_input = [1, current_z, latest_vol, latest_logvol, 0, 0]
    z_change = micro_model.predict([pred_input])[0]
    current_z = current_z + z_change
    projected_z.append(current_z)

# 4. 组合 long-term 幂律趋势 + short-term 残差推演
future_pred_df = pd.DataFrame({'Date': future_projection_dates, 'Projected_ZScore': projected_z})
genesis_date = pd.to_datetime('2009-01-09')
future_pred_df['Days'] = (future_pred_df['Date'] - genesis_date).dt.days
future_pred_df['log_Days'] = np.log(future_pred_df['Days'])

# 长期估值
future_pred_df['Base_Log_Price'] = model.predict(sm.add_constant(future_pred_df['log_Days']))
# 加权微观残差修正
future_pred_df['Hybrid_Log_Price'] = future_pred_df['Base_Log_Price'] + (future_pred_df['Projected_ZScore'] * res_std)
future_pred_df['Hybrid_Price'] = np.exp(future_pred_df['Hybrid_Log_Price'])
future_pred_df['Base_Price'] = np.exp(future_pred_df['Base_Log_Price'])

# 5. 查询今天（2026-09-11）及未来的预测结果
def query_hybrid_forecast(target_date_str):
    t_date = pd.to_datetime(target_date_str)
    row = future_pred_df[future_pred_df['Date'] == t_date]
    if len(row) == 0:
        print(f"日期 {target_date_str} 不在推演范围内")
        return
    base_p = row['Base_Price'].values[0]
    hyb_p = row['Hybrid_Price'].values[0]
    z_val = row['Projected_ZScore'].values[0]
    print(f"[{target_date_str}] 纯趋势基准价: ${base_p:,.2f} | 因子修正预测价: ${hyb_p:,.2f} | 预测 Z-Score: {z_val:.2f}")
